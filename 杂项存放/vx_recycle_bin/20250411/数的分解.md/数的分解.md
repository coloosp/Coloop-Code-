#include <iostream>
#include <string>
using namespace std;

bool istaf(int n) {
    string s = to_string(n);
    if (s.find('2') !=-1 || s.find('4')!=-1) {
        return false;
    }
    return true;
}



int main()
{
    int n = 2019;
    int count = 0;

    for (int i = 1; i < 2019; i++) {
        for (int j = i + 1; j < 2019; j++) {
            int k = 2019 - i - j;
            if (istaf(i) && istaf(j) && istaf(k)) {
                if(j<k )
                count++;
            }
        }
    }


    cout << count << endl;

    return 0;
}
