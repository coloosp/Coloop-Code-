# 特别数之和
```
#include <iostream>
#include <string>
using namespace std;

bool inqure(int n){
    string s=to_string(n);
    if(s.find('2')!=-1 ||s.find('0')!=-1 || s.find('1')!=-1  || s.find('9')!=-1 ){
      return true;
    }
    return false;



}

int main()
{
  int n;
  cin >>n;
  int result=0;
  for(int i=1;i<=n;i++){
    if(inqure(i)){
      result+=i;
    }
  }
  cout <<result <<endl;


  return 0;
}
```
